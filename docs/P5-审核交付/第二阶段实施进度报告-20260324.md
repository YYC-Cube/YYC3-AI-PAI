export function withErrorHandling<T>(
  fn: () => Promise<T>,
  onError: (error: Error) => void
): Promise<T | null> {
  return fn().catch((error: unknown) => {
    if (error instanceof Error) {
      onError(error)
    } else {
      onError(new Error(String(error)))
    }
    return null
  })
}
