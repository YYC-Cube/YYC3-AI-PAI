```typescript
// 创建统一Logger
// src/utils/logger.ts
export enum LogLevel {
  DEBUG = 0,
  INFO = 1,
  WARN = 2,
  ERROR = 3,
  FATAL = 4,
}

export interface LogEntry {
  timestamp: number
  level: LogLevel
  category: string
  message: string
  data?: unknown
  stack?: string
}

export class Logger {
  private static instance: Logger | null = null
  private static isDevelopment = process.env.NODE_ENV === 'development'
  private static isProduction = process.env.NODE_ENV === 'production'
  
  private static logs: LogEntry[] = []
  private static maxLogs = 500
  private static subscribers: Set<(entry: LogEntry) => void> = new Set()
  
  // 初始化
  static init() {
    if (Logger.instance) return
    
    Logger.instance = new Logger()
    
    // 生产环境：上报错误到服务
    if (Logger.isProduction) {
      Logger.setupErrorReporting()
    }
    
    // 开发环境：控制台彩色输出
    if (Logger.isDevelopment) {
      Logger.setupConsoleStyles()
    }
  }
  
  // 日志记录
  debug(category: string, message: string, data?: unknown) {
    this.log(LogLevel.DEBUG, category, message, data)
  }
  
  info(category: string, message: string, data?: unknown) {
    this.log(LogLevel.INFO, category, message, data)
  }
  
  warn(category: string, message: string, data?: unknown) {
    this.log(LogLevel.WARN, category, message, data)
  }
  
  error(category: string, message: string, error?: Error) {
    this.log(LogLevel.ERROR, category, message, error)
  }
  
  fatal(category: string, message: string, error?: Error) {
    this.log(LogLevel.FATAL, category, message, error)
  }
  
  private log(level: LogLevel, category: string, message: string, data?: unknown) {
    const entry: LogEntry = {
      timestamp: Date.now(),
      level,
      category,
      message,
      data,
      stack: data instanceof Error ? data.stack : undefined,
    }
    
    // 添加到日志队列
    Logger.logs.push(entry)
    if (Logger.logs.length > Logger.maxLogs) {
      Logger.logs.shift()
    }
    
    // 通知订阅者
    Logger.subscribers.forEach(subscriber => subscriber(entry))
    
    // 控制台输出
    Logger.outputToConsole(entry)
    
    // 持久化到IndexedDB
    Logger.persistLog(entry)
  }
  
  private static outputToConsole(entry: LogEntry) {
    if (!Logger.isDevelopment) return
    
    const levelColors = {
      [LogLevel.DEBUG]: 'color: #888',
      [LogLevel.INFO]: 'color: #007acc',
      [LogLevel.WARN]: 'color: #ffa500',
      [LogLevel.ERROR]: 'color: #f44336',
      [LogLevel.FATAL]: 'color: #d32f2f',
    }
    
    const timestamp = new Date(entry.timestamp).toLocaleTimeString()
    const levelName = LogLevel[entry.level]
    
    console.log(
      `%c[${timestamp}] [${levelName}] [${entry.category}]`,
      levelColors[entry.level],
      entry.message
    )
    
    if (entry.data) {
      console.log(entry.data)
    }
    if (entry.stack) {
      console.error(entry.stack)
    }
  }
  
  private static persistLog(entry: LogEntry) {
    // 使用IndexedDB持久化日志
    // ...
  }
  
  private static setupErrorReporting() {
    // 上报错误到远程日志服务
    window.addEventListener('error', (event) => {
      Logger.fatal('Global', 'Uncaught error', event.error)
    })
    
    window.addEventListener('unhandledrejection', (event) => {
      Logger.fatal('Promise', 'Unhandled rejection', event.reason)
    })
  }
  
  // 订阅日志
  static subscribe(callback: (entry: LogEntry) => void) {
    Logger.subscribers.add(callback)
    return () => Logger.subscribers.delete(callback)
  }
  
  // 获取日志历史
  static getLogs(level?: LogLevel): LogEntry[] {
    if (level !== undefined) {
      return Logger.logs.filter(log => log.level >= level)
    }
    return [...Logger.logs]
  }
  
  // 清空日志
  static clearLogs() {
    Logger.logs = []
  }
}

// 使用示例
// 在App.tsx中初始化
Logger.init()

// 在组件中使用

Logger.warn('Network', 'Slow connection detected', { latency: 1500 })
Logger.error('API', 'Failed to fetch data', error)
```

                {activeTerminal.currentDirectory} $
              </span>
              <input
                value={command}
                onChange={(e) => setCommand(e.target.value)}
                onKeyPress={handleKeyPress}
                autoFocus
              />
            </div>
          </>
        )}
      </div>
    </div>
  )
}
