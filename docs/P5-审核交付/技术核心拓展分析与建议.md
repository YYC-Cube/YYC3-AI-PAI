```typescript
interface OfflineOperation {
  id: string
  type: 'insert' | 'delete' | 'format'
  timestamp: number
  data: unknown
  syncStatus: 'pending' | 'synced' | 'conflict'
}

class OfflineQueue {
  private queue: OfflineOperation[] = []
  
  enqueue(op: OfflineOperation) {
    this.queue.push(op)
    // 保存到IndexedDB
    this.persistToStorage()
  }
  
  async sync() {
    for (const op of this.queue) {
      try {
        await this.sendToServer(op)
        op.syncStatus = 'synced'
      } catch {
        op.syncStatus = 'conflict'
      }
    }
    this.queue = this.queue.filter(op => op.syncStatus !== 'synced')
  }
}
```

```typescript
interface OperationSnapshot {

class OperationHistory {
  private history: OperationSnapshot[] = []
  
  record(op: CRDTOperation, before: string, after: string) {
    this.history.push({
      before,
      after,
      operation: op,
      timestamp: Date.now()
    })
  }
  
  rollback(opId: string): string {
    const snapshot = this.history.find(s => s.operation.id === opId)
    if (snapshot) {
      return snapshot.before
    }
    throw new Error('Operation not found')
  }
```

```typescript
class MemoryMonitor {
  private samples: { timestamp: number, usedJSHeapSize: number }[] = []
  
  startSampling(interval = 5000) {
    setInterval(() => {
      if (performance.memory) {
        this.samples.push({
          timestamp: Date.now(),
          usedJSHeapSize: performance.memory.usedJSHeapSize
        })
        
        // 保留最近100个样本
        if (this.samples.length > 100) {
          this.samples.shift()
        }
        
        // 检查内存泄漏
        this.checkMemoryLeak()
      }
    }, interval)
  }
  
  private checkMemoryLeak() {
    const recent = this.samples.slice(-20)
    const trend = this.calculateTrend(recent)
    
    if (trend > 0.1) { // 每秒增长超过10%
      this.reportMemoryLeak(trend)
    }
  }
  
  private calculateTrend(samples: typeof this.samples): number {
    // 计算线性回归斜率
    const n = samples.length
    let sumX = 0, sumY = 0, sumXY = 0, sumXX = 0
    
    samples.forEach((sample, i) => {
      sumX += i
      sumY += sample.usedJSHeapSize
      sumXY += i * sample.usedJSHeapSize
      sumXX += i * i
    })
    
    const slope = (n * sumXY - sumX * sumY) / (n * sumXX - sumX * sumX)
    return slope
  }
}
```

2. **变异测试（Mutation Testing）**
```typescript
interface MutationTestResult {
  originalCode: string
  mutants: Mutant[]
  killedMutants: number
  survivedMutants: number
  mutationScore: number
}

interface Mutant {
  id: string
  type: string  // 'replace-operator' | 'delete-statement' | 'change-logic'
  position: CodePosition
  mutatedCode: string
  killed: boolean
  killingTest: string | null
}

class MutationTester {
  async runMutationTests(
    code: string,
    tests: TestCase[]
  ): Promise<MutationTestResult> {
    const mutants = this.generateMutants(code)
    
    for (const mutant of mutants) {
      const killed = await this.testMutant(mutant, tests)
      mutant.killed = killed
      mutant.killingTest = killed ? this.findKillingTest(mutant, tests) : null
    }
    
    const killedCount = mutants.filter(m => m.killed).length
    
    return {
      originalCode: code,
      mutants,
      killedMutants: killedCount,
      survivedMutants: mutants.length - killedCount,
      mutationScore: killedCount / mutants.length
    }
  }
  
  private generateMutants(code: string): Mutant[] {
    const mutants: Mutant[] = []
    const ast = this.parseAST(code)
    
    // 变异算子
    ast.traverse(node => {
      if (node.isBinaryExpression()) {
        // 替换运算符
        mutants.push(this.replaceOperator(node))
      }
      if (node.isIfStatement()) {
        // 删除if
        mutants.push(this.deleteIfStatement(node))
      }
    })
    
    return mutants
  }
}
```

```typescript
    func: FunctionInfo
  ): PropertyTest {

      property: (input) => {
        // 定义属性
        const result = func.execute(input)
        // 示例：输出长度不超过输入长度
        return this.lengthInvariant(input, result)
      },
      generator: () => this.generateRandomInput(func),
      examples: 100,
      failures: []
    }
  }
  
  async runPropertyTest(test: PropertyTest): Promise<void> {
    for (let i = 0; i < test.examples; i++) {
      const input = test.generator()
      const passed = test.property(input)
      
      if (!passed) {
        test.failures.push({
          input,
          reason: 'Property violation',
          counterexample: input
        })
      }
    }
  }
}
```
